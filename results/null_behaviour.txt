-- COUNT(*) counts rows; COUNT(column) skips NULLs
SELECT count(*) AS all_offerings, count(venue) AS with_venue, count(*) - count(venue) AS venue_null FROM offering;
   all_offerings | with_venue | venue_null
   4520 | 4312 | 208

-- '= NULL' is never true (it is UNKNOWN); use IS NULL
SELECT (SELECT count(*) FROM offering WHERE venue = NULL) AS equals_null, (SELECT count(*) FROM offering WHERE venue IS NULL) AS is_null;
   equals_null | is_null
   0 | 208

-- Comparisons with NULL give NULL (unknown), even NULL = NULL
SELECT NULL = NULL AS null_eq_null, NULL <> 'SJT101' AS null_ne_text, NULL IS NULL AS null_is_null;
   null_eq_null | null_ne_text | null_is_null
   NULL | NULL | True

-- Three-valued logic: unknown AND false = false, unknown OR true = true
SELECT (NULL::boolean AND false) AS unknown_and_false, (NULL::boolean OR true) AS unknown_or_true, (NULL::boolean AND true) AS unknown_and_true;
   unknown_and_false | unknown_or_true | unknown_and_true
   False | True | NULL

-- NOT IN with a NULL in the list returns nothing
SELECT room, room NOT IN (SELECT venue FROM offering) AS not_in_result FROM (VALUES ('ZZZ999')) AS v(room);
   room | not_in_result
   ZZZ999 | NULL

-- ... because 'ZZZ999' <> NULL is unknown. NOT EXISTS (or filtering NULLs) gives the intended answer
SELECT (SELECT count(*) FROM (VALUES ('ZZZ999')) AS v(room)         WHERE room NOT IN (SELECT venue FROM offering)) AS not_in, (SELECT count(*) FROM (VALUES ('ZZZ999')) AS v(room)         WHERE NOT EXISTS (SELECT 1 FROM offering o WHERE o.venue = v.room)) AS not_exists, (SELECT count(*) FROM (VALUES ('ZZZ999')) AS v(room)         WHERE room NOT IN (SELECT venue FROM offering WHERE venue IS NOT NULL)) AS not_in_no_nulls;
   not_in | not_exists | not_in_no_nulls
   0 | 1 | 1

-- UNIQUE (venue, semester, slot_id) lets many NULL-venue classes share one slot
SELECT semester, slot_id, count(*) AS classes_with_null_venue FROM offering WHERE venue IS NULL GROUP BY semester, slot_id HAVING count(*) > 1 ORDER BY classes_with_null_venue DESC, slot_id LIMIT 5;
   semester | slot_id | classes_with_null_venue
   FALL2026 | T06 | 11
   FALL2026 | T11 | 11
   FALL2026 | T14 | 11
   FALL2026 | T02 | 10
   FALL2026 | T07 | 10

-- GROUP BY puts every NULL venue into ONE group, although NULL = NULL is not true
SELECT venue, count(*) AS classes FROM offering WHERE slot_id = 'T05' GROUP BY venue ORDER BY count(*) DESC, venue LIMIT 3;
   venue | classes
   NULL | 7
   MB101 | 1
   MB102 | 1

-- COALESCE replaces NULL for display
SELECT offering_id, venue, COALESCE(venue, 'not allotted') AS shown_as FROM offering WHERE venue IS NULL ORDER BY offering_id LIMIT 3;
   offering_id | venue | shown_as
   60 | NULL | not allotted
   63 | NULL | not allotted
   100 | NULL | not allotted
